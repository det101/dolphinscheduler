# Code Developer Skill

## Overview

This skill develops code based on implementation plans generated by the issue-analyzer skill. It supports Spring Boot + Vue.js technology stack.

## Prerequisites

- Java 8+ and Maven installed
- Node.js and npm installed
- Git configured
- GitHub token: `export GH_TOKEN=your_token`

## Helper Functions

```bash
# Get project root
function get_project_root() {
  echo "${PROJECT_ROOT:-$(pwd)}"
}

# Create feature branch
function create_feature_branch() {
  local branch_name=$1
  local base_branch=${2:-main}
  
  git checkout "$base_branch"
  git pull origin "$base_branch"
  git checkout -b "$branch_name"
  echo "✅ Created feature branch: $branch_name"
}

# Initialize Spring Boot project
function init_spring_boot() {
  local project_name=$1
  local package_name=${2:-com.example.$project_name}
  
  mkdir -p "$project_name/src/main/java/$package_name"
  mkdir -p "$project_name/src/main/resources"
  mkdir -p "$project_name/src/test/java/$package_name"
  
  # Create pom.xml
  cat > "$project_name/pom.xml" << 'POM'
<?xml version="1.0" encoding="UTF-8"?>
<project xmlns="http://maven.apache.org/POM/4.0.0"
         xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
         xsi:schemaLocation="http://maven.apache.org/POM/4.0.0 http://maven.apache.org/xsd/maven-4.0.0.xsd">
    <modelVersion>4.0.0</modelVersion>
    
    <parent>
        <groupId>org.springframework.boot</groupId>
        <artifactId>spring-boot-starter-parent</artifactId>
        <version>2.7.18</version>
        <relativePath/>
    </parent>
    
    <groupId>com.example</groupId>
    <artifactId>demo</artifactId>
    <version>1.0.0-SNAPSHOT</version>
    <packaging>jar</packaging>
    
    <properties>
        <java.version>1.8</java.version>
    </properties>
    
    <dependencies>
        <dependency>
            <groupId>org.springframework.boot</groupId>
            <artifactId>spring-boot-starter-web</artifactId>
        </dependency>
        <dependency>
            <groupId>org.springframework.boot</groupId>
            <artifactId>spring-boot-starter-test</artifactId>
            <scope>test</scope>
        </dependency>
    </dependencies>
</project>
POM
  
  echo "✅ Initialized Spring Boot project: $project_name"
}

# Initialize Vue.js project
function init_vue_project() {
  local project_name=$1
  
  cd "$project_name" || return 1
  npm init -y
  npm install vue@3 vue-router@4 pinia axios
  npm install -D vite @vitejs/plugin-vue
  
  # Create basic structure
  mkdir -p src/components src/views src/router src/api
  mkdir -p public
  
  # Create vite.config.js
  cat > vite.config.js << 'VITE'
import { defineConfig } from 'vite'
import vue from '@vitejs/plugin-vue'

export default defineConfig({
  plugins: [vue()],
  server: {
    port: 3000,
    proxy: {
      '/api': {
        target: 'http://localhost:8080',
        changeOrigin: true
      }
    }
  }
})
VITE
  
  echo "✅ Initialized Vue.js project: $project_name"
}

# Generate REST API controller
function generate_controller() {
  local entity_name=$1
  local package_path=$2
  
  cat > "src/main/java/$package_path/controller/${entity_name}Controller.java" << 'CTRL'
package com.example.demo.controller;

import com.example.demo.entity.$entityName;
import com.example.demo.service.$entityNameService;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

import java.util.List;

@RestController
@RequestMapping("/api/$entityName")
public class $entityNameController {

    @Autowired
    private $entityNameService $entityNameService;

    @GetMapping
    public List<$entityName> getAll() {
        return $entityNameService.findAll();
    }

    @GetMapping("/{id}")
    public ResponseEntity<$entityName> getById(@PathVariable Long id) {
        return $entityNameService.findById(id)
                .map(ResponseEntity::ok)
                .orElse(ResponseEntity.notFound().build());
    }

    @PostMapping
    public $entityName create(@RequestBody $entityName entity) {
        return $entityNameService.save(entity);
    }

    @PutMapping("/{id}")
    public ResponseEntity<$entityName> update(@PathVariable Long id, @RequestBody $entityName entity) {
        return $entityNameService.findById(id)
                .map(existing -> {
                    entity.setId(id);
                    return ResponseEntity.ok($entityNameService.save(entity));
                })
                .orElse(ResponseEntity.notFound().build());
    }

    @DeleteMapping("/{id}")
    public ResponseEntity<Void> delete(@PathVariable Long id) {
        return $entityNameService.findById(id)
                .map(entity -> {
                    $entityNameService.delete(id);
                    return ResponseEntity.ok().<Void>build();
                })
                .orElse(ResponseEntity.notFound().build());
    }
}
CTRL
  
  echo "✅ Generated controller: $entityNameController"
}

# Generate service layer
function generate_service() {
  local entity_name=$1
  local package_path=$2
  
  cat > "src/main/java/$package_path/service/$entityNameService.java" << 'SRV'
package com.example.demo.service;

import com.example.demo.entity.$entityName;
import com.example.demo.repository.$entityNameRepository;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;

import java.util.List;
import java.util.Optional;

@Service
public class $entityNameService {

    @Autowired
    private $entityNameRepository $entityNameRepository;

    public List<$entityName> findAll() {
        return $entityNameRepository.findAll();
    }

    public Optional<$entityName> findById(Long id) {
        return $entityNameRepository.findById(id);
    }

    public $entityName save($entityName entity) {
        return $entityNameRepository.save(entity);
    }

    public void delete(Long id) {
        $entityNameRepository.deleteById(id);
    }
}
SRV
  
  echo "✅ Generated service: $entityNameService"
}

# Generate entity
function generate_entity() {
  local entity_name=$1
  local package_path=$2
  
  cat > "src/main/java/$package_path/entity/$entityName.java" << 'ENT'
package com.example.demo.entity;

import javax.persistence.*;
import java.time.LocalDateTime;

@Entity
@Table(name = "$entity_name")
public class $entityName {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    private String name;

    private String description;

    @Column(name = "created_at")
    private LocalDateTime createdAt;

    @Column(name = "updated_at")
    private LocalDateTime updatedAt;

    @PrePersist
    protected void onCreate() {
        createdAt = LocalDateTime.now();
        updatedAt = LocalDateTime.now();
    }

    @PreUpdate
    protected void onUpdate() {
        updatedAt = LocalDateTime.now();
    }

    // Getters and Setters
    public Long getId() { return id; }
    public void setId(Long id) { this.id = id; }
    
    public String getName() { return name; }
    public void setName(String name) { this.name = name; }
    
    public String getDescription() { return description; }
    public void setDescription(String description) { this.description = description; }
    
    public LocalDateTime getCreatedAt() { return createdAt; }
    public void setCreatedAt(LocalDateTime createdAt) { this.createdAt = createdAt; }
    
    public LocalDateTime getUpdatedAt() { return updatedAt; }
    public void setUpdatedAt(LocalDateTime updatedAt) { this.updatedAt = updatedAt; }
}
ENT
  
  echo "✅ Generated entity: $entityName"
}

# Generate Vue component
function generate_vue_component() {
  local component_name=$1
  
  cat > "src/components/$component_name.vue" << 'VUE'
<template>
  <div class="$component_name">
    <h2>{{ title }}</h2>
    <slot></slot>
  </div>
</template>

<script>
export default {
  name: '$componentName',
  props: {
    title: {
      type: String,
      default: 'Title'
    }
  }
}
</script>

<style scoped>
.$component_name {
  padding: 20px;
}
</style>
VUE
  
  echo "✅ Generated Vue component: $component_name"
}

# Run tests
function run_tests() {
  cd "$(get_project_root)" || return 1
  
  echo "🧪 Running backend tests..."
  mvn test
  
  echo "🧪 Running frontend tests..."
  npm run test
  
  echo "✅ All tests completed"
}

# Commit changes
function commit_changes() {
  local message=$1
  
  git add .
  git commit -m "$message"
  echo "✅ Committed: $message"
}

# Push to remote
function push_changes() {
  local branch=$1
  
  git push -u origin "$branch"
  echo "✅ Pushed to: $branch"
}
```

## Commands

### 1. Develop based on implementation plan

```bash
# develop <issue_number> <repo> <implementation_plan>
function develop() {
  local issue_number=$1
  local repo=$2
  local plan=$3
  
  echo "🚀 Starting development for issue #$issue_number..."
  
  # Create feature branch
  local branch_name="feature/issue-$issue_number"
  create_feature_branch "$branch_name"
  
  # Initialize project structure
  init_spring_boot "backend" "com.example.issue$issue_number"
  init_vue_project "frontend"
  
  # Generate code based on plan
  # This would parse the implementation plan and generate appropriate code
  echo "📝 Generating code based on implementation plan..."
  
  # Run tests
  run_tests
  
  # Commit changes
  commit_changes "feat: implement issue #$issue_number"
  
  # Push to remote
  push_changes "$branch_name"
  
  echo "✅ Development completed for issue #$issue_number"
}
```

### 2. Generate specific component

```bash
# generate_component <type> <name> [params...]
# type: entity, controller, service, vue
function generate_component() {
  local type=$1
  local name=$2
  shift 2
  
  case $type in
    entity)
      generate_entity "$name" "com.example.demo"
      ;;
    controller)
      generate_controller "$name" "com.example.demo"
      ;;
    service)
      generate_service "$name" "com.example.demo"
      ;;
    vue)
      generate_vue_component "$name"
      ;;
    *)
      echo "Unknown component type: $type"
      return 1
      ;;
  esac
}
```

## Best Practices

1. Keep commits atomic and descriptive
2. Write tests before or alongside code
3. Follow existing code style and conventions
4. Update documentation as you develop
5. Run full test suite before committing
